function getTodayStr() { const now = new Date(); const y = now.getFullYear(); const m = String(now.getMonth() + 1).padStart(2, "0"); const d = String(now.getDate()).padStart(2, "0"); return `${y}-${m}-${d}`; } function render() { renderTabs(); const todayStr = getTodayStr(); // 1. 탭에 따른 목록 필터링 ('전체 일정' 탭이면 오늘 날짜 일정만 표시) const displayTasks = current === "전체 일정" ? data.tasks.filter(t => t.date === todayStr) : data.tasks.filter(t => t.subject === current); // 2. 통계 카드용 기준 데이터 ('전체 일정' 탭은 전체 과목 기준 통계 유지) const statsBase = current === "전체 일정" ? data.tasks : displayTasks; const sorted = [...displayTasks].sort((a, b) => { if (!a.date) return 1; if (!b.date) return -1; return a.date.localeCompare(b.date); }); const today = new Date(); today.setHours(0, 0, 0, 0); const weekEnd = new Date(today); weekEnd.setDate(today.getDate() + 7); const pending = statsBase.filter(t => !t.done); document.getElementById("todoCount").textContent = pending.length; document.getElementById("doneCount").textContent = statsBase.filter(t => t.done).length; document.getElementById("weekCount").textContent = pending.filter(t => { if (!t.date) return false; const d = new Date(t.date + "T00:00:00"); return d >= today && d <= weekEnd; }).length; // 3. 상단 큰 제목, 부제목, 목록 제목 변경 document.getElementById("pageTitle").textContent = current === "전체 일정" ? "오늘 해야 하는 일정" : current; document.getElementById("subtitle").textContent = current === "전체 일정" ? `${todayStr} 기준 오늘 마감되는 일정입니다.` : current + " 관련 과제와 일정을 기록하세요."; document.getElementById("listTitle").textContent = current === "전체 일정" ? "오늘의 일정 목록" : "과목 일정 및 메모"; const list = document.getElementById("taskList"); list.innerHTML = ""; if (!sorted.length) { list.innerHTML = current === "전체 일정" ? `
오늘 해야 하는 일정이 없습니다. 🎉
` : `
아직 기록된 일정이 없습니다.
＋ 일정 추가를 눌러 첫 기록을 남겨보세요.
`; return; } sorted.forEach(task => { const item = document.createElement("div"); item.className = "task" + (task.done ? " done" : ""); const check = document.createElement("input"); check.type = "checkbox"; check.checked = task.done; check.onchange = () => toggleTask(task.id); const content = document.createElement("div"); content.className = "task-content"; const title = document.createElement("div"); title.className = "task-title"; title.textContent = task.title; const desc = document.createElement("div"); desc.className = "task-desc"; desc.textContent = task.description; const meta = document.createElement("div"); meta.className = "task-meta"; const tag = document.createElement("span"); tag.className = "tag"; tag.textContent = task.type; meta.appendChild(tag); if (current === "전체 일정") { const subject = document.createElement("span"); subject.textContent = task.subject; meta.appendChild(subject); } if (task.date) { const date = document.createElement("span"); date.textContent = "📅 " + task.date; meta.appendChild(date); } content.append(title, desc, meta); const del = document.createElement("button"); del.className = "delete"; del.textContent = "×"; del.title = "삭제"; del.onclick = () => deleteTask(task.id); item.append(check, content, del); list.appendChild(item); }); }
